Inicio 
    Definir nombre_producto como Texto;
    Definir costo_fabricacion, pocentaje_ganancia_esperada, valor_ganancia, subtotal, valor_iva, precio_de_venta como decimal;
    Definir IVA,DIVISION_PORCENTAJES como decimal;

    PORCENTAJE_IVA = 0.19;
    DIVISION_PORCENTAJES = 100;

    Escribir "Digite el nombre del producto";
    Leer nombre_producto;

    Escribir "Digite el costo de fabricacion";
    Leer costo_fabricacion;

    Escribir "Digite el porcentaje de ganancia esperado";
    Leer pocentaje_ganancia_esperada;

    valor_ganancia <- (costo_fabricacion*pocentaje_ganancia_esperada)/DIVISION_PORCENTAJES;
    subtotal <- costo_fabricacion + valor_ganancia;
    valor_iva <- subtotal*PORCENTAJE_IVA;
    precio_de_venta <- subtotal + valor_iva;

    Escribir "Nombre del producto:" + nombre_producto;
    Escribir "Producto sin iva": + subtotal;
    Escribir "Valor de iva": + valor_iva;
    Escribir "Producto despues de iva": + precio_de_venta;
Fin
